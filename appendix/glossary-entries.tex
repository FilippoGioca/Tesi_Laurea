% Acronyms
\newacronym[description={Generalized Autoregressive Conditional Heteroskedasticity. Modello econometrico in cui la varianza condizionata dipende sia dai propri valori passati sia dai quadrati degli shock precedenti}]{garch}{GARCH}{Generalized Autoregressive Conditional Heteroskedasticity}

\newacronym[description={Autoregressive Conditional Heteroskedasticity. Modello in cui la varianza condizionata è funzione esclusivamente dei quadrati degli shock passati, introdotto per modellare il volatility clustering}]{arch}{ARCH}{Autoregressive Conditional Heteroskedasticity}

\newacronym[description={Glosten-Jagannathan-Runkle GARCH. Estensione asimmetrica del modello GARCH che utilizza una variabile indicatrice per catturare l'impatto maggiore degli shock negativi sulla volatilità (effetto leva)}]{gjr-garch}{GJR-GARCH}{Glosten-Jagannathan-Runkle GARCH}

\newacronym[description={Exponential Generalized Autoregressive Conditional Heteroskedasticity. Modello a volatilità condizionata specificato in logaritmi, che garantisce la positività della varianza senza imporre vincoli ai parametri e modella le risposte asimmetriche}]{egarch}{EGARCH}{Exponential Generalized Autoregressive Conditional Heteroskedasticity}

\newacronym[description={Augmented Dickey-Fuller. Test statistico impiegato per verificare la presenza di una radice unitaria in una serie storica, al fine di determinarne la stazionarietà}]{adf}{ADF}{Augmented Dickey-Fuller}

\newacronym[description={Value at Risk. Misura statistica che quantifica la perdita potenziale massima di un asset o di un portafoglio in un dato orizzonte temporale, associata a un livello di confidenza prestabilito}]{var}{VaR}{Value at Risk}

\newacronym[description={News Impact Curve. Strumento grafico che relaziona l'entità e il segno di uno shock isolato con la previsione della volatilità futura, evidenziando le risposte asimmetriche dei modelli}]{nic}{NIC}{News Impact Curve}

\newacronym[description={Autoregressive. Modello per serie storiche in cui il valore corrente della variabile è espresso come combinazione lineare dei suoi valori passati}]{ar}{AR}{Autoregressive}

\newacronym[description={Autoregressive Moving Average. Modello lineare stazionario che combina una componente autoregressiva (AR) e una componente a media mobile (MA)}]{arma}{ARMA}{Autoregressive Moving Average}

\newacronym[description={Indipendenti e Identicamente Distribuite. Proprietà di un campione in cui le variabili casuali estratte condividono la stessa distribuzione di probabilità e sono mutuamente indipendenti}]{iid}{i.i.d.}{Indipendenti e Identicamente Distribuite}

% Glossary entries
\newglossaryentry{scale-free} {
    name=scale-free,
    text=scale-free,
    sort=scale-free,
    description={in ambito finanziario ed econometrico, una misura si definisce \emph{scale-free} (letteralmente \enquote{priva di scala}) quando il suo valore è indipendente dalla scala di grandezza del capitale investito e dal prezzo nominale dell'asset. I rendimenti godono di questa proprietà poiché esprimono una variazione relativa (o percentuale), permettendo di confrontare direttamente la performance di strumenti finanziari o portafogli che presentano valori assoluti profondamente diversi tra loro}
}
